# Class: MxCADUiPrPointTransform

[2d](../modules/2d.md).MxCADUiPrPointTransform

MxCADUiPrPointTransform is an interactive point taking class that adds coordinate system transformation capabilities to MxCADUiPrPoint.

**`Description`**

This class is suitable for interactive point taking in specific coordinate systems, such as local coordinate systems, working coordinate systems, or custom view coordinate systems.
It can automatically apply matrix transformation when users select points, facilitating interactive input in the transformed space.
Usage:
1. New MxCADUiPrPointTransform (mat) creates a point picker with coordinate transformation;
2. Pass in McGeMatrix3d for mapping;
3. Call go() to obtain the user's click point;
4. The retrieved points will be transformed into the target coordinate system according to the matrix convention.

**`Example`**

```ts
   import { MxCpp, McGePoint3d, MxCADUiPrPointTransform, McGeMatrix3d, McGeVector3d, McDbPolyline } from "mxcad";

   async function Mx_Test_Rect() {
   const mxcad = MxCpp.getCurrentMxCAD();
   let mat = new McGeMatrix3d;
   mat.setToRotation(-mxcad.mxdraw.getViewAngle(), McGeVector3d.kZAxis, McGePoint3d.kOrigin);

   const getPoint = new MxCADUiPrPointTransform(mat);
   getPoint.clearLastInputPoint();
   getPoint.setDisableDynInput(true);
GetPoint.setMessage ("Please select the first point");
   const pt1 = await getPoint.go();
   if (!pt1) return;

   const getNextPt = new MxCADUiPrPointTransform(mat);
GetNextPt.setMessage ("Please select the second point");
   getNextPt.setUserDraw((pt2, pWorldDraw) => {
       let pl = new McDbPolyline();
       pl.addVertexAt(pt1);
       pl.addVertexAt(new McGePoint3d(pt1.x, pt2.y, pt1.z));
       pl.addVertexAt(pt2);
       pl.addVertexAt(new McGePoint3d(pt2.x, pt1.y, pt1.z));
       pl.isClosed = true;
       pWorldDraw.drawMcDbEntity(pl);
   });

   if (!await getNextPt.go()) return;

   getNextPt.drawReserve();
   }
```

//Example of MxCADUiPrPoint class
  import { MxCADUiPrPoint } from 'mxcad';
  import { DynamicInputType } from "mxdraw";

  async function Mx_WindowZoom() {
    let getPoint = new MxCADUiPrPointTransform(MxInsidePublicFuns.getMxCADViewSpinMatrix(MxCpp.getCurrentMxCAD()));
    getPoint.disableAllTrace();
GetPoint.setMessage ("\ n Click to zoom area");
    getPoint.setDynamicInputType(DynamicInputType.kDynTip);
    let pt1 = await getPoint.go();
    if (pt1 == null) return;

    });

    let pt2 = await getPoint.go();
    if (pt2 == null) return;

    pt1.transformBy(getPoint.rmat);
    pt2.transformBy(getPoint.rmat);
    mxcad.zoomW(pt1, pt2);
 }
