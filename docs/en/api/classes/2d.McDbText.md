# Class: McDbText

[2d](../modules/2d.md).McDbText

McDBText is a text entity class used to create and manage single line text objects.

**`Description`**

This class is used to add single line text in drawings and supports setting text content, height, position, rotation angle, alignment, and text style.

Summary of usage:
1. First, create an instance of McDBText.
2. Set the text content, position, rotation angle, alignment, and text style again.
3. Finally, insert it into the current space and update the display.

**`Example`**

```ts
//Draw vertical center aligned text pt (McGePoint3d) text position point
import { MxCpp, McDbText } from "mxcad"

const text = new McDbText(); //  Constructing Text Entity Objects
Text.textString="Test Text";//Set Object Content
text.horizontalMode = McDb.TextHorzMode.kTextMid; //  Set text alignment method
text.alignmentPoint = text.position = pt; //  Set text alignment points and text positions
MxCpp.getCurrentMxCAD().drawEntity(text); //  Draw text objects

```
```ts
//Align by Line: Select text objects and align them in the direction of the target line
  import { MxCADResbuf, MxCADUiPrEntity } from "mxcad";

  async function Mx_AlignByLine() {
//Select target line object
       let filter = new MxCADResbuf();
       filter.AddMcDbEntityTypes("LINE");
       const getLineObj = new MxCADUiPrEntity();
GetLineObj.setMessage ("Please select the target alignment line object (choose a straight line)");
       getLineObj.setFilter(filter);
       const lineObj_id = await getLineObj.go();
       if (!lineObj_id.id) return;
       const lineObj = lineObj_id.getMcDbEntity() as McDbLine;
       const getTextObj = new MxCADUiPrEntity();
//Select target text object
       let filter2 = new MxCADResbuf();
       filter2.AddMcDbEntityTypes("TEXT");
GetTextObj.setMessage ("Please select the target text object");
       getTextObj.setFilter(filter2);
       const textObj_id = await getTextObj.go();
       if (!textObj_id.id) return;
       const textObj = textObj_id.getMcDbEntity() as McDbText;
//Calculate the angle between a straight line and a text object
const line_v = lineObj.getFirstDeriv(lineObj.getStartPoint().val);//  Tangent vector at the initial point of a straight line
let angle1 = line_v.val.angleTo2(McGeVector3d.kXAxis, McGeVector3d.kNegateZAxis);// Angle between straight line and horizontal direction
let angle2 = textObj.rotation;// Initial angle of text
TextObj. rotate (textObj. position, angle1-angle 2)//Rotate the text object
       const mxcad = MxCpp.App.getCurrentMxCAD();
Mxcad. updateDisplay ()//Update Display
  }
```

### explode

▸ **explode**(): [`MxCADResbuf`](2d.MxCADResbuf.md)

Break the object and return the linked list of the object's data after being broken

#### Returns

[`MxCADResbuf`](2d.MxCADResbuf.md)

Rebuf data

```ts
import { McDbEntity, MxCADResbuf } from "mxcad";
//Obtain the target object
let getEnt = new MxCADUiPrEntity();
GetEnt.setMessage ("Select object to break: ");
let id = await getEnt.go();
let ent:McDbEntity = id.getMcDbEntity();
if (ent === null) return;
//Smash objects
      let retExplode: MxCADResbuf = ent.explode();
      if (retExplode.GetCount() == 0) return;
      let iExplodeConut = retExplode.GetCount();
      for (let j = 0; j < iExplodeConut; j++) {
          let tmpobj = retExplode.AtObject(j).val;
          if(tmpobj instanceof McDbEntity ){
              mxcad.drawEntity(tmpobj);
          }
      }
```
